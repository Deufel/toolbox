package main

import "fmt"

// Greeter holds a name.
type Greeter struct {
	name string
}

func (g *Greeter) Hello() string {
	msg := `hello, ` + g.name
	ch := make(chan int, 1)
	ch <- 42
	return fmt.Sprintf("%s (%d)", msg, <-ch)
}
