# Reactive kernel — DAG-based notebook
    from typing import Any, Callable
    import ast
    
    class Cell:
        """A single notebook cell with code + computed value."""
    
        def __init__(self, src: str, deps: list[str] = None):
            self.src = src
            self.deps = deps or []
            self.value: Any = None
    
        @property
        def is_dirty(self) -> bool:
            return self.value is None
    
        async def compute(self, env: dict) -> Any:
            tree = ast.parse(self.src, mode='exec')
            exec(compile(tree, '<cell>', 'exec'), env)
            return env.get('_')

// Datastar-style signal binding example
    const signals = new Map();
    
    class Signal {
      #value;
      #subscribers = new Set();
    
      constructor(initial = null) {
        this.#value = initial;
      }
    
      get value() { return this.#value; }
    
      set value(next) {
        if (next === this.#value) return;
        this.#value = next;
        for (const fn of this.#subscribers) fn(next);
      }
    
      subscribe(fn) {
        this.#subscribers.add(fn);
        return () => this.#subscribers.delete(fn);
      }
    }

// Package main — hub with debounced SSE drain
    package main
    
    import (
      "context"
      "fmt"
      "net/http"
      "sync"
      "time"
    )
    
    type Hub struct {
      mu       sync.RWMutex
      clients  map[string]chan []byte
      debounce time.Duration
    }
    
    // NewHub returns a fresh hub with the given debounce window.
    func NewHub(d time.Duration) *Hub {
      return &Hub{
        clients:  make(map[string]chan []byte),
        debounce: d,
      }
    }
    
    func (h *Hub) Subscribe(id string) <-chan []byte {
      h.mu.Lock()
      defer h.mu.Unlock()
      ch := make(chan []byte, 16)
      h.clients[id] = ch
      return ch
    }
    
    func (h *Hub) Publish(ctx context.Context, msg []byte) error {
      select {
      case <-ctx.Done():
        return ctx.Err()
      default:
      }
      h.mu.RLock()
      defer h.mu.RUnlock()
      for _, ch := range h.clients {
        select {
        case ch <- msg:
        default:
          fmt.Println("dropped: client buffer full")
        }
      }
      return nil
    }
    
    const (
      defaultPort = ":8080"
      maxClients  = 1_000
      pi          = 3.14159
    )
    
    func main() {
      hub := NewHub(50 * time.Millisecond)
      http.HandleFunc("/sse", func(w http.ResponseWriter, r *http.Request) {
        ch := hub.Subscribe(r.RemoteAddr)
        for msg := range ch {
          fmt.Fprintf(w, "data: %s\n\n", msg)
        }
      })
      _ = http.ListenAndServe(defaultPort, nil)
    }
